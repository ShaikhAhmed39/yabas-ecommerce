import { useEffect, useState } from 'react';
import { AuthProvider, useAuth } from '@/context/AuthContext';
import { CartProvider, useCart } from '@/context/CartContext';
import { GoldTicker } from '@/components/GoldTicker';
import { Header } from '@/components/Header';
import { Hero } from '@/components/Hero';
import { ProductGrid } from '@/components/ProductGrid';
import { About } from '@/components/About';
import { Footer } from '@/components/Footer';
import { CartDrawer } from '@/components/CartDrawer';
import { AuthModal } from '@/components/AuthModal';
import { CheckoutModal } from '@/components/CheckoutModal';
import { Lightbox } from '@/components/Lightbox';
import { AdminPage } from '@/pages/AdminPage';
import { MyOrders } from '@/pages/MyOrders';
import { ReturnRequestForm } from '@/components/ReturnRequestForm';
import { supabase } from '@/lib/supabase';
import { uploadReturnPhoto } from '@/utils/returnPhotoUpload';
import { PolicyPage, isPolicyPath } from '@/pages/PolicyPage';

type Page = 'storefront' | 'admin' | 'my-orders' | 'return-request';

type SelectedReturn = {
  orderId: string;
  orderItemId: string;

function Storefront({ onNavigate }: { onNavigate: (page: Page) => void }) {
  const { user } = useAuth();
  const { setCartOpen } = useCart();
  const [authOpen, setAuthOpen] = useState(false);
  const [checkoutOpen, setCheckoutOpen] = useState(false);

  useEffect(() => {
    supabase
      .from('products')
      .select('*')
      .order('created_at', { ascending: false })
      .then(({ data }) => {
        if (data) window.__allProducts__ = data;
      });
  }, []);

  const handleCheckout = () => {
    setCartOpen(false);
    if (!user) setAuthOpen(true);
    else setCheckoutOpen(true);
  };

  return (
    <div className="min-h-screen bg-cream-100">
      <GoldTicker />
      <Header
        onNavigate={onNavigate}
        currentPage="storefront"
        onSignIn={() => setAuthOpen(true)}
      />
      <main>
        <Hero />
        <ProductGrid />
        <About />
      </main>
      <Footer />
      <CartDrawer onCheckout={handleCheckout} />
      <AuthModal
        open={authOpen}
        onClose={() => setAuthOpen(false)}
        title="Sign In to Checkout"
      />
      <CheckoutModal
        open={checkoutOpen}
        onClose={() => setCheckoutOpen(false)}
        onRequireAuth={() => {
          setCheckoutOpen(false);
          setAuthOpen(true);
        }}
      />
      <Lightbox />
    </div>
  );
}

function AppContent() {
  const [page, setPage] = useState<Page>('storefront');
  const [selectedReturn, setSelectedReturn] = useState<SelectedReturn | null>(null);
  const [authOpen, setAuthOpen] = useState(false);
  const { user, isAdmin, loading } = useAuth();

  useEffect(() => {
    if (loading) return;

    const checkHash = () => {
      const hash = window.location.hash;

      if (hash === '#admin') {
        setPage(isAdmin ? 'admin' : 'storefront');
      } else if (hash === '#my-orders' || hash === '#return-request') {
        if (user) {
          setPage(hash === '#return-request' && selectedReturn
            ? 'return-request'
            : 'my-orders');
        } else {
          setPage('storefront');
          setAuthOpen(true);
        }
      } else {
        setPage('storefront');
      }
    };

    checkHash();
    window.addEventListener('hashchange', checkHash);
    return () => window.removeEventListener('hashchange', checkHash);
  }, [user, isAdmin, loading, selectedReturn]);

  const navigate = (newPage: Page) => {
    if (newPage === 'admin' && !isAdmin) {
      newPage = 'storefront';
    }

    if (
      (newPage === 'my-orders' || newPage === 'return-request') &&
      !user
    ) {
      setAuthOpen(true);
      return;
    }

    if (newPage === 'return-request' && !selectedReturn) {
      newPage = 'my-orders';
    }

    setPage(newPage);

    window.location.hash =
      newPage === 'admin'
        ? 'admin'
        : newPage === 'my-orders'
          ? 'my-orders'
          : newPage === 'return-request'
            ? 'return-request'
            : '';

    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const openReturnForm = (
    order: { id: string },
    item: { id: string; product_title: string },
  ) => {
    setSelectedReturn({
      orderId: order.id,
      orderItemId: item.id,
      productTitle: item.product_title,
    });
    navigate('return-request');
  };

  const renderPage: Page =
    page === 'admin' && isAdmin
      ? 'admin'
      : page === 'return-request' && user && selectedReturn
        ? 'return-request'
        : page === 'my-orders' && user
          ? 'my-orders'
          : 'storefront';

  if (isPolicyPath(window.location.pathname)) {
    return <PolicyPage />;
  }

  return (
    <CartProvider>
      {renderPage === 'admin' ? (
        <AdminPage onNavigate={navigate} />
      ) : renderPage === 'my-orders' ? (
        <>
          <Header
            onNavigate={navigate}
            currentPage="my-orders"
            onSignIn={() => setAuthOpen(true)}
          />
          <MyOrders
            onBack={() => navigate('storefront')}
            onRequestReturn={openReturnForm}
          />
          <AuthModal
            open={authOpen}
            onClose={() => setAuthOpen(false)}
            title="Sign In to View Your Orders"
          />
        </>
      ) : renderPage === 'return-request' && selectedReturn ? (
        <>
          <Header
            onNavigate={navigate}
            currentPage="my-orders"
            onSignIn={() => setAuthOpen(true)}
          />
          <ReturnRequestForm
            orderId={selectedReturn.orderId}
            orderItemId={selectedReturn.orderItemId}
            productTitle={selectedReturn.productTitle}
            onCancel={() => navigate('my-orders')}
            onSubmit={async ({ reason, description, photos }) => {
              if (!user) {
                throw new Error('Please sign in before submitting a return request.');
              }

              // Create the return request first so we have its server-generated ID.
              const { data, error } = await supabase.functions.invoke(
                'submit-return-request',
                {
                  body: {
                    order_id: selectedReturn.orderId,
                    order_item_id: selectedReturn.orderItemId,
                    reason,
                    description,
                  },
                },
              );

              if (error) {
                let message = error.message || 'Unable to submit your return request.';
                const context = error.context;

                if (context instanceof Response) {
                  try {
                    const responseBody = await context.clone().json();
                    if (typeof responseBody?.error === 'string') {
                      message = responseBody.error;
                    }
                  } catch {
                    // Keep the SDK error message if the response isn't JSON.
                  }
                }

                throw new Error(message);
              }

              const returnRequestId = data?.return_request?.id;

              if (!returnRequestId) {
                throw new Error(
                  'The server did not confirm a saved return request. Please check My Orders before trying again.',
                );
              }

              // Upload photos to the private bucket, then attach their paths.
              const uploadedPaths: string[] = [];

              try {
                for (const photo of photos) {
                  const path = await uploadReturnPhoto({
                    file: photo,
                    userId: user.id,
                    returnRequestId,
                  });

                  uploadedPaths.push(path);
                }

                if (uploadedPaths.length > 0) {
                  const { data: attachData, error: attachError } =
                    await supabase.functions.invoke('attach-return-photos', {
                      body: {
                        return_request_id: returnRequestId,
                        photo_paths: uploadedPaths,
                      },
                    });

                  if (attachError) {
                    let message =
                      attachError.message || 'Unable to attach return photos.';
                    const context = attachError.context;

                    if (context instanceof Response) {
                      try {
                        const responseBody = await context.clone().json();
                        if (typeof responseBody?.error === 'string') {
                          message = responseBody.error;
                        }
                      } catch {
                        // Keep the SDK error message if the response isn't JSON.
                      }
                    }

                    throw new Error(message);
                  }

                  if (
                    !attachData?.success ||
                    attachData.return_request_id !== returnRequestId
                  ) {
                    throw new Error('The server did not confirm the photo attachment.');
                  }
                }
              } catch (photoError) {
                console.error('Return photo processing failed:', photoError);
                throw new Error(
                  photoError instanceof Error
                    ? `Your return request was created, but photo processing failed: ${photoError.message} Please contact support before submitting another request.`
                    : 'Your return request was created, but photo processing failed. Please contact support before submitting another request.',
                );
              }

              // The form shows success only after all requested photos are attached.
            }}
          />
          <AuthModal
            open={authOpen}
            onClose={() => setAuthOpen(false)}
            title="Sign In to View Your Orders"
          />
        </>
      ) : (
        <Storefront onNavigate={navigate} />
      )}
    </CartProvider>
  );
}

function App() {
  return (
    <AuthProvider>
      <AppContent />
    </AuthProvider>
  );
}

export default App;

import { createContext, useContext, useEffect, useState, type ReactNode } from 'react';
import type { Session, User } from '@supabase/supabase-js';
import { supabase } from '@/lib/supabase';

interface AuthContextValue {
  session: Session | null;
  user: User | null;
  loading: boolean;
  isAdmin: boolean;
  signIn: (email: string, password: string) => Promise<{ error: string | null }>;
  signUp: (email: string, password: string) => Promise<{ error: string | null }>;
  signOut: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  const [isAdmin, setIsAdmin] = useState(false);

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setSession(data.session);
      setUser(data.session?.user ?? null);
      setLoading(false);
    });

    const { data: listener } = supabase.auth.onAuthStateChange((_event, newSession) => {
      setSession(newSession);
      setUser(newSession?.user ?? null);
      setLoading(false);
    });

    return () => {
      listener.subscription.unsubscribe();
    };
  }, []);

  useEffect(() => {
    async function checkAdmin() {
      if (!user) {
        setIsAdmin(false);
        return;
      }
      const { data } = await supabase
        .from('admin_users')
        .select('id')
        .eq('id', user.id)
        .maybeSingle();
      setIsAdmin(!!data);
    }
    checkAdmin();
  }, [user]);

  const signIn = async (email: string, password: string) => {
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    return { error: error?.message ?? null };
  };

  const signUp = async (email: string, password: string) => {
    const { error } = await supabase.auth.signUp({ email, password });
    return { error: error?.message ?? null };
  };

  const signOut = async () => {
    await supabase.auth.signOut();
  };

  return (
    <AuthContext.Provider value={{ session, user, loading, isAdmin, signIn, signUp, signOut }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth must be used within AuthProvider');
  return ctx;
}

import { createContext, useContext, useState, type ReactNode } from 'react';
import type { Product, CartItem } from '@/types';

interface CartContextValue {
  items: CartItem[];
  addToCart: (product: Product, qty?: number) => void;
  removeFromCart: (productId: string) => void;
  updateQuantity: (productId: string, qty: number) => void;
  clearCart: () => void;
  totalItems: number;
  totalPrice: number;
  isCartOpen: boolean;
  setCartOpen: (open: boolean) => void;
}

const CartContext = createContext<CartContextValue | undefined>(undefined);

export function CartProvider({ children }: { children: ReactNode }) {
  const [items, setItems] = useState<CartItem[]>([]);
  const [isCartOpen, setCartOpen] = useState(false);

  const addToCart = (product: Product, qty = 1) => {
    setItems((prev) => {
      const existing = prev.find((i) => i.product.id === product.id);
      if (existing) {
        return prev.map((i) =>
          i.product.id === product.id
            ? { ...i, quantity: Math.min(i.quantity + qty, product.stock) }
            : i,
        );
      }
      return [...prev, { product, quantity: Math.min(qty, product.stock) }];
    });
  };

  const removeFromCart = (productId: string) => {
    setItems((prev) => prev.filter((i) => i.product.id !== productId));
  };

  const updateQuantity = (productId: string, qty: number) => {
    setItems((prev) =>
      prev.map((i) =>
        i.product.id === productId
          ? { ...i, quantity: Math.max(1, Math.min(qty, i.product.stock)) }
          : i,
      ),
    );
  };

  const clearCart = () => setItems([]);

  const totalItems = items.reduce((sum, i) => sum + i.quantity, 0);
  const totalPrice = items.reduce((sum, i) => sum + i.product.selling_price * i.quantity, 0);

  return (
    <CartContext.Provider
      value={{
        items,
        addToCart,
        removeFromCart,
        updateQuantity,
        clearCart,
        totalItems,
        totalPrice,
        isCartOpen,
        setCartOpen,
      }}
    >
      {children}
    </CartContext.Provider>
  );
}

export function useCart() {
  const ctx = useContext(CartContext);
  if (!ctx) throw new Error('useCart must be used within CartProvider');
  return ctx;
}

async function verifyIndianPincode(pincode: string): Promise<boolean> {
  const response = await fetch(`https://api.postalpincode.in/pincode/${pincode}`);

  const file = e.target.files?.[0];

  if (file.size > 100 * 1024 * 1024) {
    setError('File too large. Maximum 100MB.');
    return;
  }

  // Allow only supported product image formats
  const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];

  if (!allowedTypes.includes(file.type)) {
    setError(
      'Invalid file type. Please upload JPG, JPEG, PNG, or WebP images only.'
    );
    return;
  }

  setUploadFile(file);
  setError(null);

  setUploadInfo({
    original: formatFileSize(file.size),
    compressed: 'Compressing...',
  });
  

    // Instantly display high performance local URL preview object
    const previewUrl = URL.createObjectURL(file);
    setImagePreview(previewUrl);

    try {
      setUploading(true);
      const compressed = await compressImageToWebP(file);
      
      setUploadInfo({
        original: formatFileSize(file.size),
        compressed: formatFileSize(compressed.size),
      });

      const fileName = `product-${Date.now()}.webp`;
      const filePath = `${fileName}`;

      const { error: uploadError } = await supabase.storage
        .from('product-images')
        .upload(filePath, compressed, {
          contentType: 'image/webp',
          upsert: false,
        });

      if (uploadError) throw uploadError;

      const { data: urlData } = supabase.storage
        .from('product-images')
        .getPublicUrl(filePath);

      setForm((prev) => ({ ...prev, image_url: urlData.publicUrl }));

export function MyOrders({
  onBack,
  onRequestReturn,
}: MyOrdersProps) {
  const { user, loading: authLoading } = useAuth();

  const [orders, setOrders] = useState<Order[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [refreshing, setRefreshing] = useState(false);

  const loadOrders = useCallback(async () => {
    if (!user) {
      setOrders([]);
      setLoading(false);
      setRefreshing(false);
      return;
    }

    setError(null);

    const { data: orderData, error: orderError } = await supabase
      .from('orders')
      .select(
        'id, user_id, total_amount, status, payment_method, payment_status, delivery_status, delivered_at, created_at',
      )
      .eq('user_id', user.id)
      .order('created_at', { ascending: false });

    if (orderError) {
      setError('We could not load your orders. Please try again.');
      setLoading(false);
      setRefreshing(false);
      return;
    }

    const fetchedOrders = (orderData || []) as Omit<Order, 'items'>[];

    if (fetchedOrders.length === 0) {
      setOrders([]);
      setLoading(false);
      setRefreshing(false);
      return;
    }

import { supabase } from '@/lib/supabase';
import { compressImageToWebP } from '@/utils/imageCompress';

const BUCKET = 'return-photos';
const MAX_OUTPUT_SIZE = 10 * 1024 * 1024; // Must match the private bucket's 10 MB limit.
const MAX_ORIGINAL_SIZE = 10 * 1024 * 1024; // 10 MB

const ALLOWED_TYPES = new Set([
  'image/jpeg',
  'image/png',
  'image/webp',
]);

/**
 * Compress and upload one return photo to the private bucket.
 *
 * Storage path:
 * userId/returnRequestId/unique-file-name.webp
 *
 * The return request ID must be a valid UUID.
 * The caller must supply the authenticated user's ID.
 */
export async function uploadReturnPhoto({
  file,
  userId,
  returnRequestId,
}: {
  file: File;
  userId: string;
  returnRequestId: string;
}): Promise<string> {
  if (!file) {
    throw new Error('Please select a photo.');
  }

  if (!ALLOWED_TYPES.has(file.type)) {
    throw new Error('Only JPEG, PNG, and WebP photos are allowed.');
  }

  if (file.size === 0 || file.size > MAX_ORIGINAL_SIZE) {
    throw new Error('The original photo must be 10 MB or smaller.');
  }

  if (!userId || !returnRequestId) {
    throw new Error('Missing user or return request information.');
  }

  // Confirm that the current session belongs to this user.
  const {
    data: { user },
    error: authError,
  } = await supabase.auth.getUser();

  if (authError || !user || user.id !== userId) {
    throw new Error('Please sign in with the correct account.');
  }

  // Reuse the existing compressor without changing product uploads.
  const compressed = await compressImageToWebP(file);

  if (compressed.size === 0 || compressed.size > MAX_OUTPUT_SIZE) {
    throw new Error(
      'The compressed photo exceeds the 10 MB upload limit. Please choose a smaller image.',
    );
  }

  const uniqueName = `${crypto.randomUUID()}.webp`;
  const path = `${userId}/${returnRequestId}/${uniqueName}`;

  const { error: uploadError } = await supabase.storage
    .from(BUCKET)
    .upload(path, compressed, {
      contentType: 'image/webp',
      cacheControl: '3600',
      upsert: false,
    });

  if (uploadError) {
    throw new Error(`Photo upload failed: ${uploadError.message}`);
  }

  // Return the private storage path, NOT a public URL.
  return path;
}

  "Access-Control-Allow-Headers":
    "authorization, apikey, content-type, x-client-info, x-supabase-api-version",

type JsonObject = Record<string, unknown>;

type CustomerDetails = {
  customer_name: string;
  shipping_address_line1: string;
  shipping_address_line2: string | null;
  city: string;
  state: string;
  pincode: string;
  primary_phone: string;
  backup_phone: string | null;
};

function response(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: {
      ...corsHeaders,
      "Content-Type": "application/json",
      "Cache-Control": "no-store",
    },
  });
}

function isRecord(value: unknown): value is JsonObject {
  return (
    typeof value === "object" &&
    value !== null &&
    !Array.isArray(value)
  );
}

function cleanString(value: unknown): string | null {
  if (typeof value !== "string") return null;

  const cleaned = value.trim();
  return cleaned.length > 0 ? cleaned : null;
}

function isUuid(value: unknown): value is string {
  return (
    typeof value === "string" &&
    /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(
      value,
    )
  );
}

function validateCustomer(value: unknown):
  | { customer: CustomerDetails }
  | { error: string } {
  if (!isRecord(value)) {
    return { error: "Invalid customer details" };
  }

  const customerName = cleanString(value.customer_name);
  const address1 = cleanString(value.shipping_address_line1);
  const city = cleanString(value.city);
  const state = cleanString(value.state);
  const pincode = cleanString(value.pincode);
  const phone = cleanString(value.primary_phone);

  if (!customerName || customerName.length > 120) {
    return { error: "Invalid customer name" };
  }

  if (!address1 || address1.length > 300) {
    return { error: "Invalid delivery address" };
  }

  if (!city || city.length > 100) {
    return { error: "Invalid city" };
  }

  if (!state || state.length > 100) {
    return { error: "Invalid state" };
  }

  if (!pincode || !/^\d{6}$/.test(pincode)) {
    return { error: "Pincode must contain exactly 6 digits" };
  }

  if (!phone || !/^\d{10}$/.test(phone)) {
    return { error: "Phone number must contain exactly 10 digits" };
  }

  const address2 = value.shipping_address_line2;
  const backupPhone = value.backup_phone;

  if (
    address2 !== undefined &&
    address2 !== null &&
    typeof address2 !== "string"
  ) {
    return { error: "Invalid second address line" };
  }

  if (
    backupPhone !== undefined &&
    backupPhone !== null &&
    typeof backupPhone !== "string"
  ) {
    return { error: "Invalid backup phone number" };
  }

  const normalizedBackupPhone =
    typeof backupPhone === "string" && backupPhone.trim()
      ? backupPhone.trim()
      : null;

  if (
    normalizedBackupPhone !== null &&
    !/^\d{10}$/.test(normalizedBackupPhone)
  ) {
    return { error: "Backup phone must contain exactly 10 digits" };
  }

  return {
    customer: {
      customer_name: customerName,
      shipping_address_line1: address1,
      shipping_address_line2:
        typeof address2 === "string" ? address2.trim() || null : null,
      city,
      state,
      pincode,
      primary_phone: phone,
      backup_phone: normalizedBackupPhone,
    },
  };
}

function paymentResponse(
  order: {
    id: string;
    total_amount: number | string;
    status: string;
    payment_status: string;
    razorpay_order_id: string | null;
  },
  keyId: string,
) {
  return response({
    orderId: order.id,
    razorpayOrderId: order.razorpay_order_id,
    amount: Math.round(Number(order.total_amount) * 100),
    currency: "INR",
    keyId,
  });
}

Deno.serve(async (req: Request): Promise<Response> => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  if (req.method !== "POST") {
    return response({ error: "Method not allowed" }, 405);
  }

  try {
    const supabaseUrl = Deno.env.get("SUPABASE_URL");
    const anonKey = Deno.env.get("SUPABASE_ANON_KEY");
    const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
    const razorpayKeyId = Deno.env.get("RAZORPAY_KEY_ID");
    const razorpayKeySecret = Deno.env.get("RAZORPAY_KEY_SECRET");

    if (
      !supabaseUrl ||
      !anonKey ||
      !serviceKey ||
      !razorpayKeyId ||
      !razorpayKeySecret
    ) {
      console.error("Required payment configuration is missing");
      return response({ error: "Payment service is not configured" }, 503);
    }

    const authorization = req.headers.get("Authorization");

    if (!authorization?.startsWith("Bearer ")) {
      return response({ error: "Please sign in before checkout" }, 401);
    }

    const token = authorization.slice("Bearer ".length).trim();

    if (!token) {
      return response({ error: "Invalid authentication token" }, 401);
    }

    // Validate the customer's Supabase access token.
    const userClient = createClient(supabaseUrl, anonKey, {
      auth: {
        persistSession: false,
        autoRefreshToken: false,
      },
      global: {
        headers: { Authorization: `Bearer ${token}` },
      },
    });

    const { data: userData, error: authError } =
      await userClient.auth.getUser(token);

    if (authError || !userData.user) {
      return response({ error: "Your session is invalid or expired" }, 401);
    }

    // Privileged operations are performed only after authentication.
    const admin = createClient(supabaseUrl, serviceKey, {
      auth: {
        persistSession: false,
        autoRefreshToken: false,
      },
    });

    let body: unknown;

    try {
      body = await req.json();
    } catch {
      return response({ error: "Request body must be valid JSON" }, 400);
    }

    if (!isRecord(body)) {
      return response({ error: "Invalid checkout request" }, 400);
    }

    const { items, customer, checkout_request_id } = body;

    if (!isUuid(checkout_request_id)) {
      return response(
        { error: "A valid checkout_request_id is required" },
        400,
      );
    }

    const customerResult = validateCustomer(customer);

    if ("error" in customerResult) {
      return response({ error: customerResult.error }, 400);
    }

    if (
      !Array.isArray(items) ||
      items.length === 0 ||
      items.length > 50
    ) {
      return response({ error: "Invalid cart items" }, 400);
    }

    // Combine duplicate product IDs and reject malformed quantities.
    const quantities = new Map<string, number>();

    for (const item of items) {
      if (!isRecord(item) || !isUuid(item.product_id)) {
        return response({ error: "Invalid product in cart" }, 400);
      }

      if (
        !Number.isSafeInteger(item.quantity) ||
        Number(item.quantity) < 1 ||
        Number(item.quantity) > 20
      ) {
        return response({ error: "Invalid product quantity" }, 400);
      }

      const productId = item.product_id;
      const newQuantity =
        (quantities.get(productId) ?? 0) + Number(item.quantity);

      if (newQuantity > 20) {
        return response(
          { error: "Maximum quantity per product is 20" },
          400,
        );
      }

      quantities.set(productId, newQuantity);
    }

    // Fast-path safe retries. Legacy orders without a recorded reservation
    // cannot be reused by the hardened checkout flow.
    const { data: existing, error: existingError } = await admin
      .from("orders")
      .select(
        "id, user_id, total_amount, status, payment_status, razorpay_order_id, inventory_status",
      )
      .eq("user_id", userData.user.id)
      .eq("checkout_request_id", checkout_request_id)
      .maybeSingle();

    if (existingError) {
      console.error("Existing checkout lookup failed:", existingError);
      return response({ error: "Could not check existing checkout" }, 500);
    }

    if (existing) {
      if (existing.status !== "pending" || existing.payment_status !== "pending") {
        return response({ error: "This checkout is no longer pending" }, 409);
      }
      if (existing.inventory_status !== "reserved") {
        return response(
          { error: "This checkout needs reconciliation before it can continue." },
          409,
        );
      }
      if (existing.razorpay_order_id) {
        return paymentResponse(existing, razorpayKeyId);
      }
      return response(
        { error: "This checkout is already being processed. Please wait while it is reconciled." },
        409,
      );
    }

    // Reserve stock and create the internal order + item snapshots in one
    // database transaction. Prices and stock are read by the trusted RPC.
    const customerDetails = customerResult.customer;
    const rpcItems = [...quantities.entries()].map(([product_id, quantity]) => ({
      product_id,
      quantity,
    }));

    const { data: reservationData, error: reservationError } = await admin.rpc(
      "create_order_with_inventory_reservation",
      {
        p_user_id: userData.user.id,
        p_checkout_request_id: checkout_request_id,
        p_customer: customerDetails,
        p_items: rpcItems,
      },
    );

    if (reservationError || !isRecord(reservationData)) {
      const message = reservationError?.message ?? "Invalid inventory reservation response";
      console.error("Atomic inventory reservation failed:", message);

      if (/insufficient stock|product unavailable|stock reservation failed/i.test(message)) {
        return response({ error: "One or more products are out of stock. Please refresh your cart." }, 409);
      }
      if (/invalid customer|invalid product|invalid checkout|invalid line amount|minimum online payment/i.test(message)) {
        return response({ error: "Checkout details or cart are invalid. Please review and try again." }, 400);
      }
      return response({ error: "Could not reserve inventory for checkout. Please retry shortly." }, 500);
    }

    const orderId = reservationData.orderId;
    const totalAmount = Number(reservationData.totalAmount);
    const reservedInventory = reservationData.inventoryStatus === "reserved";
    const isExistingReservation = reservationData.existing === true;
    const existingRazorpayOrderId =
      typeof reservationData.razorpayOrderId === "string"
        ? reservationData.razorpayOrderId
        : null;

    if (
      typeof orderId !== "string" ||
      !Number.isFinite(totalAmount) ||
      totalAmount <= 0 ||
      !reservedInventory
    ) {
      console.error("Inventory RPC returned an unexpected result");
      return response({ error: "Checkout needs reconciliation before payment can continue." }, 500);
    }

    // A race may cause the RPC to return an order created by another request.
    // Reuse it only if its Razorpay order ID is already persisted. Never create
    // a second remote payment order for an uncertain prior request.
    if (isExistingReservation) {
      if (existingRazorpayOrderId) {
        return paymentResponse({
          id: orderId,
          total_amount: totalAmount,
          status: "pending",
          payment_status: "pending",
          razorpay_order_id: existingRazorpayOrderId,
        }, razorpayKeyId);
      }
      return response(
        { error: "This checkout is already being processed. Please wait while it is reconciled." },
        409,
      );
    }

    const totalPaise = Math.round(totalAmount * 100);
    if (!Number.isSafeInteger(totalPaise) || totalPaise < 100) {
      console.error("Inventory RPC returned an invalid total amount");
      return response({ error: "Invalid order amount. Please contact support." }, 500);
    }

    let razorpayOrder: {
      id?: string;
      amount?: number;
      currency?: string;
      receipt?: string;
      status?: string;
    };

    try {
      const razorpayResponse = await fetch(
        "https://api.razorpay.com/v1/orders",
        {
          method: "POST",
          headers: {
            Authorization:
              `Basic ${btoa(`${razorpayKeyId}:${razorpayKeySecret}`)}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            amount: totalPaise,
            currency: "INR",
            receipt: orderId,
          }),
          signal: AbortSignal.timeout(15000),
        },
      );

      if (!razorpayResponse.ok) {
        const errorText = await razorpayResponse.text();
        const status = razorpayResponse.status;

        console.error(
          "Razorpay order creation returned an error:",
          status,
          errorText.slice(0, 1000),
        );

        // Only a clear client-side rejection is treated as definitive. A
        // timeout, rate limit, conflict, or server error may be ambiguous;
        // keep stock reserved until reconciliation in those cases.
        const definitiveRejection =
          status >= 400 && status < 500 && ![408, 409, 429].includes(status);

        if (definitiveRejection) {
          const { error: failError } = await admin
            .from("orders")
            .update({
              payment_status: "failed",
              payment_error: "Razorpay definitively rejected order creation",
            })
            .eq("id", orderId)
            .is("razorpay_order_id", null);

          if (failError) {
            console.error("Could not mark rejected checkout failed:", failError);
          } else {
            const { error: releaseError } = await admin.rpc(
              "release_failed_order_inventory",
              { p_order_id: orderId },
            );
            if (releaseError) {
              console.error("Could not release stock after definitive rejection:", releaseError);
            }
          }
        } else {
          const { error: recordError } = await admin
            .from("orders")
            .update({ payment_error: "Razorpay order creation outcome needs reconciliation" })
            .eq("id", orderId);
          if (recordError) {
            console.error("Could not record payment setup error:", recordError);
          }
        }

        return response(
          {
            error: definitiveRejection
              ? "Razorpay rejected payment setup. Please start a new checkout attempt."
              : "Payment setup needs reconciliation. Please wait or contact support before retrying.",
          },
          502,
        );
      }

      razorpayOrder = await razorpayResponse.json();

      if (
        typeof razorpayOrder.id !== "string" ||
        razorpayOrder.amount !== totalPaise ||
        razorpayOrder.currency !== "INR"
      ) {
        console.error("Unexpected Razorpay order response");

        await admin
          .from("orders")
          .update({
            payment_error: "Unexpected Razorpay order response",
          })
          .eq("id", orderId);

        return response(
          {
            error:
              "Payment setup needs reconciliation. Please contact support before retrying.",
          },
          502,
        );
      }
    } catch (error) {
      // A timeout or network failure does not prove Razorpay failed to create
      // its order. Keep the internal record and do not blindly retry creation.
      console.error("Razorpay request outcome is uncertain:", error);

      await admin
        .from("orders")
        .update({
          payment_error: "Razorpay creation outcome is uncertain",
        })
        .eq("id", orderId);

      return response(
        {
          error:
            "We could not confirm payment setup. Please wait or contact support before retrying.",
        },
        502,
      );
    }

    const { data: updatedOrder, error: updateError } = await admin
      .from("orders")
      .update({
        razorpay_order_id: razorpayOrder.id,
        payment_error: null,
      })
      .eq("id", orderId)
      .eq("user_id", userData.user.id)
      .select(
        "id, user_id, total_amount, status, payment_status, razorpay_order_id",
      )
      .single();

    if (
      updateError ||
      !updatedOrder ||
      updatedOrder.razorpay_order_id !== razorpayOrder.id
    ) {
      console.error("Saving Razorpay order reference failed:", updateError);

      // The Razorpay order may already exist. Preserve the internal order
      // for reconciliation; never delete it or create another blindly.
      await admin
        .from("orders")
        .update({
          payment_error: "Could not save Razorpay order reference",
        })
        .eq("id", orderId);

      return response(
        {
          error:
            "Payment setup needs reconciliation. Please contact support before retrying.",
        },
        500,
      );
    }

    return paymentResponse(updatedOrder, razorpayKeyId);
  } catch (error) {
    console.error("Unexpected checkout function error:", error);
    return response({ error: "Could not process checkout" }, 500);
  }
});

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null &&
    !Array.isArray(value);

function safeEqualHex(a: string, b: string): boolean {
  if (!/^[a-f0-9]{64}$/i.test(a) || !/^[a-f0-9]{64}$/i.test(b)) {
    return false;
  }

  const left = a.toLowerCase();
  const right = b.toLowerCase();

  let difference = 0;
  for (let i = 0; i < left.length; i++) {
    difference |= left.charCodeAt(i) ^ right.charCodeAt(i);
  }

  return difference === 0;
}

async function createHmacHex(
  secret: string,
  message: string,
): Promise<string> {
  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );

  const signature = await crypto.subtle.sign(
    "HMAC",
    key,
    new TextEncoder().encode(message),
  );

  return Array.from(new Uint8Array(signature))
    .map((byte) => byte.toString(16).padStart(2, "0"))
    .join("");
}

      !/^pay_[A-Za-z0-9]+$/.test(razorpayPaymentId) ||

        400,

      auth: {
        persistSession: false,
        autoRefreshToken: false,
      },

    const razorpayResponse = await fetch(

      {

        { error: "Could not confirm payment status with Razorpay. Please retry." },

      );
    }

      payment_status: "paid",

function jsonResponse(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: {
      "Content-Type": "application/json",
      "Cache-Control": "no-store",
    },
  });
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

async function hmacHex(secret: string, body: string): Promise<string> {
  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );

  const signature = await crypto.subtle.sign(
    "HMAC",
    key,
    new TextEncoder().encode(body),
  );

  return Array.from(new Uint8Array(signature))
    .map((byte) => byte.toString(16).padStart(2, "0"))
    .join("");
}

function safeEqualHex(a: string, b: string): boolean {
  if (!/^[a-f0-9]{64}$/i.test(a) || !/^[a-f0-9]{64}$/i.test(b)) {
    return false;
  }

  const left = a.toLowerCase();
  const right = b.toLowerCase();
  let difference = 0;

  for (let i = 0; i < left.length; i++) {
    difference |= left.charCodeAt(i) ^ right.charCodeAt(i);
  }

  return difference === 0;
}

function isRazorpayId(
  value: unknown,
  prefix: "order_" | "pay_",
): value is string {
  return typeof value === "string" &&
    value.length <= 100 &&
    new RegExp(`^${prefix}[A-Za-z0-9]+$`).test(value);
}

  if (!safeEqualHex(expectedSignature, receivedSignature)) {
    return jsonResponse({ error: "Invalid webhook signature." }, 401);

        !Number.isSafeInteger(expectedAmountPaise) ||
        expectedAmountPaise <= 0 ||

      ) {
        console.error("Webhook amount does not match order.", {
          orderId: order.id,
          razorpayOrderId,
        });
        return jsonResponse({ error: "Payment amount mismatch." }, 400);
      }

        "confirm_order_payment_and_commit_inventory",
        {
          p_order_id: order.id,
          p_razorpay_payment_id: paymentId,

        confirmation.paymentStatus !== "paid" ||

    return response;

      .eq("id", requestId)
      .maybeSingle();

    if (requestError) {
      console.error("Return lookup failed:", requestError.message);
      return respond({ error: "Unable to load the return request." }, 500);
    }

    if (!request) {
      return respond({ error: "Return request not found." }, 404);
    }

    if (!transition.from.includes(request.status)) {
      return respond(
        {
          error: `Cannot perform '${action}' when the request status is '${request.status}'.`,
        },
        409,
      );
    }

    const updates: Record<string, unknown> = {
      status: transition.to,
      updated_at: new Date().toISOString(),
    };

    if (adminNote) {
      updates.admin_note = adminNote;
    }

    if (action === "reject") {
      updates.rejection_reason = rejectionReason;
    }

    if (action === "schedule_pickup") {
      updates.pickup_scheduled_at = pickupScheduledAt;
    }

    if (action === "mark_refund_pending") {
      if (
        typeof body.refund_amount !== "number" ||
        !Number.isFinite(body.refund_amount) ||
        body.refund_amount <= 0
      ) {
        return respond(
          { error: "Provide a valid positive refund amount." },
          400,
        );
      }

      const { data: item, error: itemError } = await db
        .from("order_items")
        .select("unit_price, quantity")
        .eq("id", request.order_item_id)
        .eq("order_id", request.order_id)
        .maybeSingle();

      if (itemError || !item) {
        console.error("Return item verification failed.");
        return respond({ error: "Unable to verify the returned item." }, 500);
      }

      const itemTotal =
        Math.round(Number(item.unit_price) * Number(item.quantity) * 100) / 100;

      if (
        !Number.isFinite(itemTotal) ||
        body.refund_amount > itemTotal
      ) {
        return respond(
          { error: "Refund amount cannot exceed this item's original total." },
          400,
        );
      }

      updates.refund_amount =
        Math.round(body.refund_amount * 100) / 100;
    }

    // Never permit this function to claim that money has been refunded.
    // Actual refund initiation must use a separately verified payment workflow.
    const { data: updated, error: updateError } = await db
      .from("return_requests")
      .update(updates)
      .eq("id", requestId)
      .eq("status", request.status)
      .select("id, status, admin_note, rejection_reason, pickup_scheduled_at, refund_amount, updated_at")
      .maybeSingle();

    if (updateError) {
      console.error("Return update failed:", updateError.message);
      return respond({ error: "Unable to update the return request." }, 500);
    }

    if (!updated) {
      return respond(
        {
          error:
            "The request changed during this operation. Refresh and try again.",
        },
        409,
      );
    }

    return respond({
      message: "Return request updated successfully.",
      return_request: updated,
    });
  } catch (error) {
    console.error(
      "Unexpected return-management error:",
      error instanceof Error ? error.message : "Unknown error",
    );

    return respond({ error: "An unexpected server error occurred." }, 500);
  }
});

import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",

const ALLOWED_STATUSES = [
  "pending",
  "processing",
  "shipped",
  "out_for_delivery",
  "delivered",
  "cancelled",
] as const;

Deno.serve(async (req: Request) => {
  // Catch option pre-flight requests from frontend
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  if (req.method !== "POST") {
    return new Response(
      JSON.stringify({ error: "Method not allowed." }),
      { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }

  try {
    const authHeader = req.headers.get("Authorization");
    if (!authHeader?.startsWith("Bearer ")) {
      return new Response(
        JSON.stringify({ error: "Authentication required." }),
        { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    const token = authHeader.slice("Bearer ".length);
    const supabaseUrl = Deno.env.get("SUPABASE_URL");
    const anonKey = Deno.env.get("SUPABASE_ANON_KEY");
    const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

    if (!supabaseUrl || !anonKey || !serviceRoleKey) {
      throw new Error("Server configuration is incomplete.");
    }

    // Verify the user's JWT with Supabase Auth.
    const authClient = createClient(supabaseUrl, anonKey, {
      auth: {
        autoRefreshToken: false,
        persistSession: false,
      },
    });

    const { data: authData, error: authError } =
      await authClient.auth.getUser(token);

    if (authError || !authData.user) {
      return new Response(
        JSON.stringify({ error: "Invalid or expired session." }),
        { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Use the service role only after authenticating the caller.
    const adminClient = createClient(supabaseUrl, serviceRoleKey, {
      auth: {
        autoRefreshToken: false,
        persistSession: false,
      },
    });

    // Confirm the caller is in the trusted admin allowlist.
    const { data: admin, error: adminError } = await adminClient
      .from("admin_users")
      .select("id")
      .eq("id", authData.user.id)
      .maybeSingle();

    if (adminError) {
      console.error("Admin authorization lookup failed:", adminError);
      return new Response(
        JSON.stringify({ error: "Could not verify administrator access." }),
        { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    if (!admin) {
      return new Response(
        JSON.stringify({ error: "Administrator access required." }),
        { status: 403, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    const body = await req.json();
    const orderId = body?.order_id;
    const nextStatus = body?.delivery_status;

    if (
      typeof orderId !== "string" ||
      !/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(orderId)
    ) {
      return new Response(
        JSON.stringify({ error: "A valid order ID is required." }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    if (
      typeof nextStatus !== "string" ||
      !ALLOWED_STATUSES.includes(
        nextStatus as (typeof ALLOWED_STATUSES)[number],
      )
    ) {
      return new Response(
        JSON.stringify({ error: "Invalid delivery status." }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Record a delivery timestamp only when marking the order delivered.
    const updatePayload = {
      delivery_status: nextStatus,
      status:nextStatus,
      delivered_at:
        nextStatus === "delivered" ? new Date().toISOString() : null,
    };

    const { data: order, error: updateError } = await adminClient
      .from("orders")
      .update(updatePayload)
      .eq("id", orderId)
      .select("id, delivery_status, delivered_at")
      .maybeSingle();

    if (updateError) {
      console.error("Delivery update failed:", updateError);
      return new Response(
        JSON.stringify({ error: "Could not update delivery status." }),
        { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    if (!order) {
      return new Response(
        JSON.stringify({ error: "Order not found." }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    return new Response(
      JSON.stringify({ success: true, order }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  } catch (error) {
    console.error("Unexpected delivery function error:", error);
    return new Response(
      JSON.stringify({ error: "Unexpected server error." }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }
});

import { createClient } from "npm:@supabase/supabase-js@2";

Deno.serve(async (req: Request) => {
  if (req.headers.get("x-cron-secret") !== Deno.env.get("CRON_SECRET")) {
    return new Response("Unauthorized", { status: 401 });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  );

  const res = await fetch("https://www.goldapi.io/api/price/XAU/INR", {
    headers: { "x-access-token": Deno.env.get("GOLDAPI_KEY")! },
  });
  if (!res.ok) return new Response(`GoldAPI error ${res.status}`, { status: 502 });

  const data = await res.json();
  const raw = Number(data?.melt_price_per_gram?.["24k"]);
  if (!Number.isFinite(raw) || raw < 1000 || raw > 100000) {
    return new Response("Invalid price from API", { status: 502 });
  }

  const { data: setting } = await supabase
    .from("site_settings").select("value")
    .eq("key", "gold_margin_percent").single();
  const margin = Number(setting?.value ?? 0);
  const finalRate = Math.round(raw * (1 + margin / 100));

  const { error } = await supabase.from("gold_rates").insert({
    rate_per_gram_24k: finalRate,
    raw_rate_per_gram: raw,
    margin_percent: margin,
  });
  if (error) return new Response(error.message, { status: 500 });

  return Response.json({ ok: true, raw, margin, finalRate });
});
